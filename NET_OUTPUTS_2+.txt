Program 10: Web Application to Insert, Update, and Delete Records
using System;


using System.Data;


using System.Data.OleDb;


public partial class _Default : System.Web.UI.Page


{


    string ConString = @"Provider=Microsoft.ACE.OLEDB.12.0;Data Source=D:\EmpDB.accdb";


    protected void Page_Load(object sender, EventArgs e)


    {


        if (!IsPostBack)


        {


            DisplayRecords();


        }


    }


    private void DisplayRecords()


    {


        string query = "SELECT * FROM Emptable";


        using (OleDbConnection con = new OleDbConnection(ConString))


        {


            using (OleDbDataAdapter adapter = new OleDbDataAdapter(query, con))


            {


                DataTable dt = new DataTable();


                con.Open();


                adapter.Fill(dt);


                GridView1.DataSource = dt;


                GridView1.DataBind();


            }


        }


    }


    // Insert Record


    protected void Button1_Click(object sender, EventArgs e)


    {


        string query = "INSERT INTO Emptable (Id, EName, Did, DName, Salary) VALUES (?, ?, ?, ?, ?)";


        using (OleDbConnection con = new OleDbConnection(ConString))


        {


            using (OleDbCommand cmd = new OleDbCommand(query, con))


            {


                cmd.Parameters.AddWithValue("@Id", TextBox1.Text);


                cmd.Parameters.AddWithValue("@EName", TextBox2.Text);


                cmd.Parameters.AddWithValue("@Did", TextBox3.Text);


                cmd.Parameters.AddWithValue("@DName", TextBox4.Text);


                cmd.Parameters.AddWithValue("@Salary", TextBox5.Text);


                con.Open();


                cmd.ExecuteNonQuery();


            }


        }


        Label1.Text = "Record inserted successfully";


        DisplayRecords();


        


        TextBox1.Text = "";


        TextBox2.Text = "";


        TextBox3.Text = "";


        TextBox4.Text = "";


        TextBox5.Text = "";


    }


    // Update Record


    protected void Button2_Click(object sender, EventArgs e)


    {


        string query = "UPDATE Emptable SET salary = salary * 1.15 WHERE id = ?";


        using (OleDbConnection con = new OleDbConnection(ConString))


        {


            using (OleDbCommand cmd = new OleDbCommand(query, con))


            {


                cmd.Parameters.AddWithValue("@Id", TextBox1.Text);


                con.Open();


                int result = cmd.ExecuteNonQuery();


                if (result > 0)


                {


                    Label1.Text = "Salary increased by 15% successfully";


                }


                else


                {


                    Label1.Text = "Employee not found";


                }


            }


        }


        DisplayRecords();


        TextBox1.Text = "";


    }


    // Delete Record


    protected void Button3_Click(object sender, EventArgs e)


    {


        string query = "DELETE FROM Emptable WHERE id = ?";


        using (OleDbConnection con = new OleDbConnection(ConString))


        {


            using (OleDbCommand cmd = new OleDbCommand(query, con))


            {


                cmd.Parameters.AddWithValue("@Id", TextBox1.Text);


                con.Open();


                int result = cmd.ExecuteNonQuery();


                if (result > 0)


                {


                    Label1.Text = "Record deleted successfully";


                }


                else


                {


                    Label1.Text = "Employee not found";


                }


            }


        }


        DisplayRecords();


        TextBox1.Text = "";


    }


}

Program 12: Implement the Data Reader Class
using System;


using System.Data;


using System.Data.OleDb;


using System.Web.UI.WebControls;


public partial class _Default : System.Web.UI.Page


{


    protected void Page_Load(object sender, EventArgs e)


    {


        if (!IsPostBack)


        {


            LoadEmployeeData();


        }


    }


    private void LoadEmployeeData()


    {


        string connectionString = @"Provider=Microsoft.ACE.OLEDB.12.0;Data Source=D:\EmpDB.accdb;";


        string query = "SELECT * FROM Emptable";


        using (OleDbConnection con = new OleDbConnection(connectionString))


        {


            using (OleDbCommand cmd = new OleDbCommand(query, con))


            {


                con.Open();


                using (OleDbDataReader reader = cmd.ExecuteReader())


                {


                    GridView1.DataSource = reader;


                    GridView1.DataBind();


                }


            }


        }


    }
