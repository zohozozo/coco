private void btnDIS_Click(object sender, EventArgs e)

{

int num;

num = Convert.ToInt32(txtNUM.Text);

string result = "Square :\n";

for(int i = 1; i <= num; i++)

{

result += i + "^2 = " + (i * i) + "\n";

}

int sum = 0;

int j = 1;

while (j <= num)

{

sum += j;

j++;

}

result += "\nSum of Natural Numbers =" + sum;

lblRES.Text = result;

}

For all text box validating:

private void textBox1_Validating(object sender, CancelEventArgs e)

{

if (textBox1.Text.Trim() == "")

{

errorProvider1.SetError(textBox1, "Enter Mark");

e.Cancel = true;

return;

}

int mark = Convert.ToInt32(textBox1.Text);

if (mark < 0 || mark > 100)

{

errorProvider1.SetError(textBox1, "Mark should be " +

"between 0 and 100");

e.Cancel = true;

}

else

{

errorProvider1.SetError(textBox1, "");

}

}

private void button1_Click(object sender, EventArgs e)

{

int m1 = Convert.ToInt32(textBox1.Text);

int m2 = Convert.ToInt32(textBox2.Text);

int m3 = Convert.ToInt32(textBox3.Text);

int total = m1 + m2 + m3;

double average = total / 3.0;

string result;

if (m1 >= 35 && m2 >= 35 && m3 >= 35)

result = "Pass";

else

result = "Fail";

this.Hide();

Form2 frm = new Form2(total, average, result);

frm.ShowDialog();

Application.Exit();
